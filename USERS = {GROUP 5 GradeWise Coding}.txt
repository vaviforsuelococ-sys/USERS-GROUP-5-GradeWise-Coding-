def get_valid_mark(subject_name):
    """Validates mark inputs (0-100) using try-except error handling. Allows 'back' to cancel."""
    while True:
        val = input(f"Enter Marks In {subject_name} (0-100) [Type 'back' to return]: ").strip()
        if val.lower() == 'back':
            return None
        try:
            mark = float(val)
            if 0 <= mark <= 100:
                return mark
            else:
                print("Invalid input. Please enter a score between 0 and 100.")
        except ValueError:
            print("Invalid input. Please enter a valid numerical value.")

def calculate_grades(student_name, section, period):
    """Computes total, average, percentage, and grading status for a specific period."""
    print(f"\n==========================================")
    print(f" Recording {period} Grades for: {student_name} ({section})")
    print(f" Focus Major Subjects: Introduction to Computing & Computer Programming")
    print(f"==========================================")
    
    marks = {}
    
    if period == "P1":
        subjects = [
            "PATHFIT 1", "Purposive Communication", "Understanding the Self",
            "Ethics", "Mathematics in the Modern World", "NSTP 1",
            "Introduction to Computing [MAJOR]", "Computer Programming 1 [MAJOR]"
        ]
    elif period == "P2":
        subjects = [
            "PATHFIT 2", "Art Appreciation", "Readings in Philippine History",
            "Science, Technology, and Society", "Introduction to Computing 2 [MAJOR]",
            "Computer Programming 2 [MAJOR]"
        ]
    else: # P3
        subjects = [
            "Physical Education 3", "Contemporary World",
            "Data Structures and Algorithms [MAJOR]", "Object-Oriented Programming [MAJOR]"
        ]
        
    for subj in subjects:
        mark = get_valid_mark(subj)
        if mark is None:
            print("\n[Cancelled] Grade recording aborted. Returning to menu.")
            return
        marks[subj] = mark
        
    total = sum(marks.values())
    count = len(marks)
    average = total / count
    percentage = (total / (count * 100)) * 100
    
    if percentage >= 90:
        status = "Passed (Excellent)"
    elif percentage >= 80:
        status = "Passed (Very Satisfactory)"
    elif percentage >= 70:
        status = "Passed"
    else:
        status = "Failed"
        
    if student_name not in GRADES_DATA:
        GRADES_DATA[student_name] = {"section": section, "periods": {}}
        
    GRADES_DATA[student_name]["periods"][period] = {
        "marks": marks,
        "total": total,
        "average": average,
        "percentage": percentage,
        "status": status
    }
    
    print(f"\n[Success] Grades successfully recorded for {student_name}!")
    print(f"Total: {total:.2f} | Average: {average:.2f} | Percentage: {percentage:.2f}% | Status: {status}")

def record_grade_flow():
    """Handles section, student, and grading period selection with back navigation."""
    while True:
        print(f"\n--- Available Sections ---")
        sections = list(STUDENT_DATA.keys())
        for idx, sec in enumerate(sections, 1):
            print(f"[{idx}] {sec}")
        print("[0] Return to Teacher Menu")
            
        sec_choice = input("Select Section Number: ").strip()
        if sec_choice == "0":
            return
        if sec_choice.isdigit() and 1 <= int(sec_choice) <= len(sections):
            selected_section = sections[int(sec_choice) - 1]
            students = STUDENT_DATA[selected_section]
            
            while True:
                print(f"\n--- Student List for {selected_section} ---")
                for idx, student in enumerate(students, 1):
                    print(f"[{idx}] {student}")
                print("[0] Back to Section Selection")
                    
                st_choice = input("Select Student Number: ").strip()
                if st_choice == "0":
                    break
                if st_choice.isdigit() and 1 <= int(st_choice) <= len(students):
                    selected_student = students[int(st_choice) - 1]
                    
                    while True:
                        print(f"\n--- Select Grading Period for {selected_student} ---")
                        print("[1] P1 (Period 1)")
                        print("[2] P2 (Period 2)")
                        print("[3] P3 (Period 3)")
                        print("[0] Back to Student List")
                        period_choice = input("Select Period (0-3): ").strip()
                        
                        if period_choice == "0":
                            break
                        period_map = {"1": "P1", "2": "P2", "3": "P3"}
                        if period_choice in period_map:
                            selected_period = period_map[period_choice]
                            calculate_grades(selected_student, selected_section, selected_period)
                            return
                        else:
                            print("\n[Error] Invalid grading period selection.")
                else:
                    print("\n[Error] Invalid student selection.")
        else:
            print("\n[Error] Invalid section selection.")

            print(f"\n[Notice] No grade records found for '{student_name}'. Please check spelling or consult your instructor.")

def view_all_records():

        return
        
    for student, info in GRADES_DATA.items():

        print("=" * 80)
        for period, p_data in info["periods"].items():

            print(f"{'Subject Name':<42} | {'Grade':<6}")

            for subj, mark in p_data["marks"].items():

            print(f"Total Score: {p_data['total']:.2f} | Average: {p_data['average']:.2f} | Percentage: {p_data['percentage']:.2f}% | Status: {p_data['status']}")
        print("=" * 80)
        
    input("\nPress Enter to return to Teacher Menu...")

def teacher_menu():
    """Teacher navigation menu."""
    while True:
        print(f"\n==========================================")
        print(f"              TEACHER MENU                ")
        print(f"==========================================")
        print("[1] Record / Update Grade (P1, P2, P3)")
        print("[2] View All Student Records (Summary with Grades)")
        print("[3] Log Out / Return to Main Menu")
        choice = input("Select an option (1-3): ").strip()
        
        if choice == "1":
            record_grade_flow()
        elif choice == "2":
            view_all_records()
        elif choice == "3":
            print("\nLogged out successfully.")
            break
        else:
            print("\n[Error] Invalid selection, please try again.")

def login():
    """Main login function with loop, conditional branching, and easy undo/back navigation."""
    while True:
        print(f"\n==========================================")
        print(f"             SYSTEM LOG IN                ")
        print(f"==========================================")
        print("[1] Student Portal")
        print("[2] Teacher Portal")
        print("[3] Exit System")
        choice = input("Select User Type (1-3): ").strip()
        
        if choice == "1":
            view_student_grades_table()
        elif choice == "2":
            print(f"\n[Teacher Access Selected]")
            username = input("Username [Type 'back' to return]: ").strip()
            if username.lower() == 'back':
                continue
            password = input("Password: ").strip()
            
            if username in USERS and USERS[username] == password:
                print("\nLogin Successful!")
                teacher_menu()
            else:
                print("\n[Error] Invalid Username or Password. Returning to main menu.")
        elif choice == "3":
            print("\nExiting system. Goodbye!")
            break
        else:
            print("\n[Error] Invalid choice. Please select 1, 2, or 3.")
